\section{Milestone 1: Foundations \& Basic Behavior}
\textit{Goal: Establish the simulated environment and foundational drone control logic in 2D.}

\subsection{Suffiyan (Member A) --- Physics \& Engine}
\begin{itemize}[noitemsep]
    \item \textbf{World Setup:} Create 1000x1000 unit continuous plane with boundary repulsion logic.
    \item \textbf{Obstacles:} Place static rectangular obstacles at fixed positions.
    \item \textbf{Main Loop:} Implement the simulation loop with fixed time step $\Delta t = 0.02s$ (50 FPS target).
\end{itemize}

\subsection{Sharique (Member B) --- Swarm Logic}
\begin{itemize}[noitemsep]
    \item \textbf{Drone Class:} Define state vector (pos, vel, heading) and update kinematics.
    \item \textbf{Sensing:} Implement neighbor detection within a fixed interaction radius $R$.
    \item \textbf{Avoidance:} Initial reactive avoidance rules for drone-drone safety.
\end{itemize}

\subsection{Usman (Member C) --- Visualization \& UI}
\begin{itemize}[noitemsep]
    \item \textbf{Rendering:} Pygame-based rendering for 100 drones and obstacles.
    \item \textbf{Analytics:} Real-time HUD for FPS, drone counts, and neighbor link highlights.
    \item \textbf{Controls:} Simulation playback controls (Pause/Resume/Step).
\end{itemize}

\subsection{Asshal (Member D) --- Optimization}
\begin{itemize}[noitemsep]
    \item \textbf{Benchmarking:} Baseline $O(N^2)$ neighbor check and performance logging.
    \item \textbf{Indexing:} Implement 2D Grid Hash or Quadtree for optimized $O(N)$ lookups.
    \item \textbf{Vectorization:} Port core math to NumPy for batch-update efficiency.
\end{itemize}

\section{Milestone 2: Decentralized Coordination}

\begin{itemize}

\section{Evaluation Metrics}
\begin{itemize}[noitemsep]
    \item \textbf{Mission Completion Time:} Ticks to complete tasks.
    \item \textbf{Collision Count:} Inter-drone and drone-obstacle collision frequency.
    \item \textbf{Area Coverage:} \% of target airspace covered in time limit.
    \item \textbf{Robustness:} Performance stability under 25\% drone failure (fault injection).
\end{itemize}
